+export type RateLimiter = { check(key: string): boolean };
+
+export function createRateLimiter(options: { max: number; windowMs: number; maxEntries?: number; now?: () => number }): RateLimiter {
+  const maxEntries = options.maxEntries ?? 10_000;
+  const buckets = new Map<string, { count: number; resetAt: number }>();
+  const now = () => (options.now ?? Date.now)();
+  const sweep = (at: number) => {
+    for (const [key, bucket] of buckets) {
+      if (bucket.resetAt <= at) {
+        buckets.delete(key);
+      }
+    }
+  };
+  return {
+    check(key) {
+      const at = now();
+      if (buckets.size >= maxEntries) {
+        sweep(at);
+      }
+      const bucket = buckets.get(key);
+      if (!bucket) {
+        if (buckets.size >= maxEntries) {
+          return false;
+        }
+        buckets.set(key, { count: 1, resetAt: at + options.windowMs });
+        return true;
+      }
+      if (bucket.resetAt <= at) {
+        buckets.set(key, { count: 1, resetAt: at + options.windowMs });
+        return true;
+      }
+      if (bucket.count >= options.max) {
+        return false;
+      }
+      bucket.count += 1;
+      return true;
+    },
+  };

+type ProxyOptions = { target?: string };

+const proxy = (viteConfig.server?.proxy ?? {}) as ProxyMap;

+function proxyEntryFor(url: string): string | undefined {
+  return Object.keys(proxy).find((context) =>

import { useEffect } from 'react';
import { useQueryClient, type QueryKey } from '@tanstack/react-query';

export function useEvents(queryKeys: QueryKey[]): void {
  const queryClient = useQueryClient();

  useEffect(() => {
    if (typeof EventSource === 'undefined') {
      return;
    }

    const source = new EventSource('/api/events');
    source.onopen = () => {
      for (const queryKey of queryKeys) {
        queryClient.invalidateQueries({ queryKey });
      }
    };

    return () => {
      source.close();
    };
  }, [queryClient, queryKeys]);
}

import { useState } from 'react';
import { useQuery } from '@tanstack/react-query';
import { api } from '../../api';
import { Button, Card, EmptyState } from '../../components/ui';

type AuditEntry = {
  id: string;
  userId: string | null;
  action: string;
  targetType: string | null;
  targetId: string | null;
  ip: string | null;
  createdAt: number;
};

const PAGE_SIZE = 20;

function targetLabel(entry: AuditEntry): string {
  if (entry.targetType && entry.targetId) {
    return `${entry.targetType}/${entry.targetId}`;
  }
  return entry.targetType ?? entry.targetId ?? '—';
}

export function Audit() {
  const [page, setPage] = useState(1);
  const audit = useQuery({
    queryKey: ['audit', page],
    queryFn: () => api<{ entries: AuditEntry[]; total: number }>(`/api/audit?page=${page}&limit=${PAGE_SIZE}`),
  });

  const entries = audit.data?.entries ?? [];
  const total = audit.data?.total ?? 0;
  const totalPages = Math.max(1, Math.ceil(total / PAGE_SIZE));

  return (
    <Card title="Auditoria">
      {audit.isError ? <p className="text-sm text-red-400">Falha ao carregar a auditoria.</p> : null}
      {audit.isLoading ? <p className="text-sm text-slate-500">Carregando…</p> : null}
      {!audit.isLoading && !audit.isError && entries.length === 0 ? (
        <EmptyState>Nenhuma entrada de auditoria.</EmptyState>
      ) : null}
      {entries.length > 0 ? (
        <table className="w-full text-sm">
          <thead>
            <tr className="text-left text-xs uppercase tracking-wide text-slate-500">
              <th className="py-2 font-medium">Data</th>
              <th className="py-2 font-medium">Usuário</th>
              <th className="py-2 font-medium">Ação</th>
              <th className="py-2 font-medium">Alvo</th>
              <th className="py-2 font-medium">IP</th>
            </tr>
          </thead>
          <tbody>
            {entries.map((entry) => (
              <tr key={entry.id} className="border-t border-slate-800">
                <td className="py-2 text-slate-300">{new Date(entry.createdAt).toLocaleString('pt-BR')}</td>
                <td className="py-2 text-slate-300">{entry.userId ?? '—'}</td>
                <td className="py-2 text-slate-300">{entry.action}</td>
                <td className="py-2 text-slate-300">{targetLabel(entry)}</td>
                <td className="py-2 text-slate-300">{entry.ip ?? '—'}</td>
              </tr>
            ))}
          </tbody>
        </table>
      ) : null}
      <div className="mt-4 flex items-center justify-between gap-2">
        <Button type="button" onClick={() => setPage((current) => current - 1)} disabled={page === 1 || audit.isFetching}>
          Anterior
        </Button>
        <span className="text-sm text-slate-400">
          Página {page} de {totalPages}
        </span>
        <Button
          type="button"
          onClick={() => setPage((current) => current + 1)}
          disabled={page * PAGE_SIZE >= total || audit.isFetching}
        >
          Próxima
        </Button>
      </div>
    </Card>
  );
}

function json(data: unknown, status = 200) {
  return new Response(JSON.stringify(data), { status });
}

function renderAudit() {
  const client = new QueryClient({ defaultOptions: { queries: { retry: false } } });
  return render(
    <QueryClientProvider client={client}>
      <Audit />
    </QueryClientProvider>,
  );
}

describe('Audit', () => {
  afterEach(() => {
    vi.unstubAllGlobals();
  });

  it('lista entradas paginadas e navega para a próxima página', async () => {
    const fetchMock = vi.fn(async (input: RequestInfo | URL) => {
      const url = String(input);
      if (url === '/api/audit?page=1&limit=20') {
        return json({ entries: firstPage, total: 21 });
      }
      if (url === '/api/audit?page=2&limit=20') {
        return json({ entries: secondPage, total: 21 });
      }
      return json({ error: { code: 'not_found', message: 'Rota não encontrada' } }, 404);
    });
    vi.stubGlobal('fetch', fetchMock);
    renderAudit();

    expect(await screen.findByText('acao-1')).toBeInTheDocument();
    expect(screen.getByText('usuario-1')).toBeInTheDocument();
    expect(screen.getByText('user/t1')).toBeInTheDocument();
    expect(screen.getByText('10.0.0.1')).toBeInTheDocument();
    expect(screen.getByText(new Date(firstPage[0]!.createdAt).toLocaleString('pt-BR'))).toBeInTheDocument();
    expect(screen.getByText('Página 1 de 2')).toBeInTheDocument();
    expect(screen.getByRole('button', { name: 'Anterior' })).toBeDisabled();
    expect(screen.getByRole('button', { name: 'Próxima' })).toBeEnabled();

    await userEvent.click(screen.getByRole('button', { name: 'Próxima' }));

    expect(await screen.findByText('acao-21')).toBeInTheDocument();
    expect(screen.getByText('usuario-21')).toBeInTheDocument();
    expect(screen.getByText('user/t21')).toBeInTheDocument();
    expect(screen.getByText('10.0.0.21')).toBeInTheDocument();
    expect(screen.getByText('Página 2 de 2')).toBeInTheDocument();
    expect(screen.getByRole('button', { name: 'Anterior' })).toBeEnabled();
    expect(screen.getByRole('button', { name: 'Próxima' })).toBeDisabled();
    expect(fetchMock.mock.calls.some(([input]) => String(input) === '/api/audit?page=2&limit=20')).toBe(true);
  });
});
